cmake_minimum_required(VERSION 3.16)
project(appdimens-gtk VERSION 3.2.0 LANGUAGES C)

option(APPDIMENS_GTK_VERSION "GTK major version to build against (3 or 4)" 4)
option(APPDIMENS_BUILD_TESTS "Build tests" ON)
option(APPDIMENS_BUILD_EXAMPLES "Build examples" ON)
set(CMAKE_C_STANDARD 99)

find_package(PkgConfig REQUIRED)
if(APPDIMENS_GTK_VERSION STREQUAL "3")
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.22)
  set(AD_LIB appdimens-gtk3)
  set(AD_GTK_PC gtk+-3.0)
else()
  pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4>=4.0)
  set(AD_LIB appdimens-gtk4)
  set(AD_GTK_PC gtk4)
endif()

# Header-only core (no GTK dependency) — usable from any C/C++ toolkit.
add_library(appdimens-core INTERFACE)
target_include_directories(appdimens-core INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
add_library(AppDimens::core ALIAS appdimens-core)

add_library(${AD_LIB} src/appdimens-gtk.c)
target_link_libraries(${AD_LIB} PUBLIC appdimens-core PkgConfig::GTK m)
set_target_properties(${AD_LIB} PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 3 C_VISIBILITY_PRESET default)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${AD_LIB} PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
add_library(AppDimens::gtk ALIAS ${AD_LIB})

if(APPDIMENS_BUILD_TESTS)
  enable_testing()
  add_executable(test_core tests/test_core.c)
  target_include_directories(test_core PRIVATE tests)
  target_link_libraries(test_core appdimens-core m)
  add_test(NAME core COMMAND test_core)
  foreach(v double nolibm fixed)
    add_executable(test_core_${v} tests/test_core.c)
    target_include_directories(test_core_${v} PRIVATE tests)
    target_link_libraries(test_core_${v} appdimens-core m)
    add_test(NAME core_${v} COMMAND test_core_${v})
  endforeach()
  target_compile_definitions(test_core_double PRIVATE AD_USE_DOUBLE)
  target_compile_definitions(test_core_nolibm PRIVATE AD_NO_LIBM)
  target_compile_definitions(test_core_fixed PRIVATE AD_FIXED_POINT)
  add_executable(test_custom tests/test_custom.c)
  target_include_directories(test_custom PRIVATE tests include/appdimens)
  target_link_libraries(test_custom appdimens-core m)
  add_test(NAME custom COMMAND test_custom)
  add_executable(test_gtk tests/test_gtk.c)
  target_link_libraries(test_gtk ${AD_LIB})
  add_test(NAME gtk COMMAND test_gtk)
  set_tests_properties(gtk PROPERTIES SKIP_RETURN_CODE 77 ENVIRONMENT "NO_AT_BRIDGE=1;GTK_A11Y=none")
endif()

if(APPDIMENS_BUILD_EXAMPLES)
  add_executable(appdimens-gtk-demo examples/demo.c)
  target_link_libraries(appdimens-gtk-demo ${AD_LIB})
endif()

include(GNUInstallDirs)
install(TARGETS ${AD_LIB} appdimens-core EXPORT AppDimensGtkTargets)
install(DIRECTORY include/appdimens DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT AppDimensGtkTargets NAMESPACE AppDimens:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AppDimensGtk)
configure_file(appdimens-gtk.pc.in ${AD_LIB}.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/${AD_LIB}.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
